    const datos = {
      nombreArchivo: nombreArchivo,
      serverURL: serverURL,

  function obtenerArchivoSeleccionado() {
    try {
      const datosStr = localStorage.getItem(ARCHIVO_SELECCIONADO_KEY);
      if (!datosStr) return null;
      
      const datos = JSON.parse(datosStr);
      
      if (Date.now() > datos.expiracion) {
        localStorage.removeItem(ARCHIVO_SELECCIONADO_KEY);
        return null;
      }
      
      return {
        nombreArchivo: datos.nombreArchivo,
        serverURL: datos.serverURL
      };
    } catch (error) {
      console.error('Error al obtener archivo seleccionado:', error);
      localStorage.removeItem(ARCHIVO_SELECCIONADO_KEY);
      return null;
    }
  }

    let compileServerURL = null; // Se establecerá cuando se encuentre una URL que funcione
    
    function obtenerURLsCompilador() {
      const urlsToTry = [];
      
      const currentOrigin = window.location.origin; // Ej: http://192.168.96.91:3002
      const currentHostname = window.location.hostname; // IP o dominio
      const currentPort = window.location.port;
      const protocol = window.location.protocol; // "http:" o "https:"
      
      console.log('🔍 Detectando URLs de compilador desde:', {
        origin: currentOrigin,
        hostname: currentHostname,
        port: currentPort,
        protocol: protocol
      });
      
      if (currentHostname) {
        if (currentHostname !== 'localhost' && currentHostname !== '127.0.0.1') {
          const compileURL = `${protocol}//${currentHostname}:3000/compile`;
          urlsToTry.push(compileURL);
          console.log('🌐 URL de compilador detectada desde red:', compileURL);
        } else {
          const compileURL = 'http://localhost:3000/compile';
          urlsToTry.push(compileURL);
          console.log('🏠 Usando localhost para compilador:', compileURL);
        }
      }
      
      if (typeof getServerURLs === 'function') {
        try {
          const configUrls = getServerURLs();
          console.log('📋 URLs desde config.js:', configUrls);
          configUrls.forEach(url => {
            if (url && typeof url === 'string') {
              let compileURL = url;
              if (compileURL.includes(':')) {
                compileURL = compileURL.replace(/:(\d+)(\/|$)/, ':3000$2');
              } else {
                compileURL = compileURL.replace(/(https?:\/\/[^\/]+)(\/|$)/, '$1:3000$2');
              }
              if (!compileURL.endsWith('/compile')) {
                compileURL = compileURL.replace(/\/$/, '') + '/compile';
              }
              if (!urlsToTry.includes(compileURL)) {
                urlsToTry.push(compileURL);
                console.log('  ➕ Agregada URL desde config:', compileURL);
              }
            }
          });
        } catch (e) {
          console.warn('⚠️ Error al obtener URLs desde config.js:', e);
        }
      } else if (typeof getServerURL === 'function') {
        try {
          const configUrl = getServerURL();
          if (configUrl && typeof configUrl === 'string') {
            let compileURL = configUrl.replace(':3002', ':3000');
            if (!compileURL.endsWith('/compile')) {
              compileURL = compileURL.replace(/\/$/, '') + '/compile';
            }
            if (!urlsToTry.includes(compileURL)) {
              urlsToTry.push(compileURL);
              console.log('  ➕ Agregada URL desde getServerURL:', compileURL);
            }
          }
        } catch (e) {
          console.warn('⚠️ Error al obtener URL desde getServerURL:', e);
        }
      }
      
      if (urlsToTry.length === 0) {
        urlsToTry.push('http://localhost:3000/compile');
        console.log('⚠️ No se detectaron URLs, usando fallback localhost');
      }
      
      console.log('🔗 URLs de compilador finales a intentar:', urlsToTry);
      return urlsToTry;
    }

    function intentarCompilar(urls, index, sourceCode, input, outputContent, outputDiv, toggleBtn) {
      if (index >= urls.length) {
        outputDiv.classList.add("maximized");
        if (toggleBtn) {
          toggleBtn.textContent = "🗗";
          toggleBtn.title = "Minimizar salida";
        }
        const urlsIntentadas = urls.join('\n- ');
        outputContent.textContent = `❌ Error: No se pudo conectar a ningún servidor de compilación.\n\nURLs intentadas:\n- ${urlsIntentadas}\n\nVerifica que:\n1. El servidor esté corriendo\n2. El firewall permita conexiones en el puerto 3000\n3. Estés usando la IP correcta del servidor\n4. Ambas computadoras estén en la misma red`;
        console.error('❌ No se pudo conectar a ningún servidor de compilación después de intentar todas las URLs:', urls);
        return;
      }

      const serverURL = urls[index];
      console.log(`🔌 Intentando compilar con servidor ${index + 1}/${urls.length}: ${serverURL}`);
      
      const compileTimeout = setTimeout(() => {
        console.log(`Compilación con ${serverURL} falló por timeout, intentando siguiente URL...`);
        intentarCompilar(urls, index + 1, sourceCode, input, outputContent, outputDiv, toggleBtn);
      }, 10000); // Timeout de 10 segundos

      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 10000);
      
      fetch(serverURL, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          code: sourceCode,
          input: input
        }),
        signal: controller.signal
      })
      .then(res => {
        clearTimeout(compileTimeout);
        clearTimeout(timeoutId);
        if (!res.ok) {
          throw new Error(`HTTP error! status: ${res.status}`);
        }
        return res.json();
      })
      .then(data => {
        compileServerURL = serverURL;
        URL = serverURL;
        console.log('✅ URL de compilación establecida:', serverURL);
        
        const output = data.output || "Sin salida";
        const partes = output.split("Mensaje original de compilador:");
        if (partes.length === 2) {
          outputDiv.classList.add("maximized");
          if (toggleBtn) {
            toggleBtn.textContent = "🗗";
            toggleBtn.title = "Minimizar salida";
          }
          outputContent.innerHTML =
            `<div style="color: darkgreen; font-weight: bold; margin-bottom: 1em;">${partes[0].trim()}</div>` +
            `<pre style="color: white; background: inherit; padding: 10px; border-radius: 5px;">${partes[1].trim()}</pre>`;
        } else if (/error|Error|compilador|compilation/i.test(output)) {
          outputDiv.classList.add("maximized");
          if (toggleBtn) {
            toggleBtn.textContent = "🗗";
            toggleBtn.title = "Minimizar salida";
          }
          outputContent.textContent = output;
        } else {
          outputContent.textContent = output;
          outputDiv.style.display = 'block';
          outputContent.style.display = 'block';
        }
      })
      .catch(err => {
        clearTimeout(compileTimeout);
        clearTimeout(timeoutId);
        
        console.error(`❌ Error al compilar con ${serverURL}:`, {
          name: err.name,
          message: err.message,
          type: err.type,
          stack: err.stack
        });
        
        if (err.name === 'AbortError' || err.name === 'TypeError' || 
            err.message.includes('fetch') || err.message.includes('Failed to fetch') ||
            err.message.includes('NetworkError') || err.message.includes('network')) {
          console.log(`⚠️ Error de conexión con ${serverURL}, intentando siguiente URL...`);
          if (index + 1 < urls.length) {
            outputContent.textContent = `Intentando conectar a servidor ${index + 2}/${urls.length}...`;
          }
          intentarCompilar(urls, index + 1, sourceCode, input, outputContent, outputDiv, toggleBtn);
        } else {
          outputDiv.classList.add("maximized");
          if (toggleBtn) {
            toggleBtn.textContent = "🗗";
            toggleBtn.title = "Minimizar salida";
          }
          outputContent.textContent = `Error al ejecutar: ${err.message}\n\nURL intentada: ${serverURL}`;
        }
      });
    }
    
    function ejecutarCodigo() {
  const editorActivo = obtenerEditorActivo();
  if (!editorActivo) {
    alert('No hay editor activo');
    return;
  }

  const sourceCode = editorActivo.getValue();
  const inputArea = document.querySelector(`.input-area[data-tab-id="${pestañaActiva}"]`);
  const outputContent = document.querySelector(`.output[data-tab-id="${pestañaActiva}"] .output-content`);
  const outputDiv = document.querySelector(`.output[data-tab-id="${pestañaActiva}"]`);
  const toggleBtn = document.querySelector(`.output[data-tab-id="${pestañaActiva}"] .toggleOutputSize`);

  if (!inputArea || !outputContent || !outputDiv) {
    console.error('Elementos no encontrados:', {
      inputArea: !!inputArea,
      outputContent: !!outputContent,
      outputDiv: !!outputDiv,
      pestañaActiva: pestañaActiva
    });
    alert('Elementos del editor no encontrados. Verifica que la pestaña esté activa.');
    return;
  }

  outputDiv.style.display = 'block';
  outputContent.style.display = 'block';
  outputContent.style.visibility = 'visible';
  outputContent.style.opacity = '1';

  const input = inputArea.value;

  const cinRegex = /cin\s*>>([^;]+)/g;
  let totalEntradasEsperadas = 0;
  let match;
  while ((match = cinRegex.exec(sourceCode)) !== null) {
    const variables = match[1].split(">>").map(v => v.trim()).filter(Boolean);
    totalEntradasEsperadas += variables.length;
  }

  const entradasUsuario = input.trim().split(/\s+/).filter(e => e.length > 0);
  const totalEntradasUsuario = entradasUsuario.length;

  if (totalEntradasEsperadas > 0 && totalEntradasUsuario !== totalEntradasEsperadas) {
    alert(`⚠️ Advertencia: el código espera ${totalEntradasEsperadas} entradas, pero se ingresaron ${totalEntradasUsuario}.
Esto puede causar errores si las entradas son insuficientes o sobran.\n\nSe ejecutará igual.`);
  }

  outputContent.textContent = "Ejecutando...";
  
  const urlsCompilacionActualizadas = obtenerURLsCompilador();
  console.log('🔧 URLs de compilación actualizadas:', urlsCompilacionActualizadas);
  
  if (compileServerURL) {
    if (urlsCompilacionActualizadas.includes(compileServerURL)) {
      console.log('✅ Usando URL de compilación guardada:', compileServerURL);
      intentarCompilar([compileServerURL], 0, sourceCode, input, outputContent, outputDiv, toggleBtn);
    } else {
      console.log('⚠️ URL de compilación guardada ya no es válida, usando URLs actualizadas');
      compileServerURL = null; // Limpiar la URL guardada
      intentarCompilar(urlsCompilacionActualizadas, 0, sourceCode, input, outputContent, outputDiv, toggleBtn);
    }
  } else {
    console.log('🔍 No hay URL de compilación guardada, probando todas las URLs');
    intentarCompilar(urlsCompilacionActualizadas, 0, sourceCode, input, outputContent, outputDiv, toggleBtn);
  }
}

    function compilarConReintentos(codigo, entrada) {
      return new Promise((resolve, reject) => {
        const urlsAProbar = compileServerURL ? [compileServerURL] : urlsCompilacion;
        
        function intentarCompilarAux(urls, index) {
          if (index >= urls.length) {
            reject(new Error("No se pudo conectar a ningún servidor de compilación"));
            return;
          }

          const serverURL = urls[index];
          console.log(`Intentando compilar (validación) con servidor ${index + 1}/${urls.length}:`, serverURL);

          const controller = new AbortController();
          const timeoutId = setTimeout(() => controller.abort(), 10000);

          fetch(serverURL, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
              code: codigo,
              input: entrada
            }),
            signal: controller.signal
          })
          .then(res => {
            clearTimeout(timeoutId);
            if (!res.ok) {
              throw new Error(`HTTP error! status: ${res.status}`);
            }
            return res.json();
          })
          .then(data => {
            if (!compileServerURL) {
              compileServerURL = serverURL;
              URL = serverURL;
              console.log('✅ URL de compilación establecida (desde validación):', serverURL);
            }
            resolve(data);
          })
          .catch(err => {
            clearTimeout(timeoutId);
            if (err.name === 'AbortError' || err.name === 'TypeError' || err.message.includes('fetch') || err.message.includes('Failed to fetch')) {
              console.log(`Error con ${serverURL}:`, err.message);
              intentarCompilarAux(urls, index + 1);
            } else {
              reject(err);
            }
          });
        }

        intentarCompilarAux(urlsAProbar, 0);
      });
    }

function moverLinea(cm, direccion) {
  console.log("estoy moviendo linea");
  const rango = cm.listSelections();
  cm.operation(() => {
    rango.forEach(sel => {
      const linea = sel.head.line;
      const otraLinea = linea + direccion;
      if (otraLinea < 0 || otraLinea >= cm.lineCount()) return;

      const textoActual = cm.getLine(linea);
      const textoOtra = cm.getLine(otraLinea);

      cm.replaceRange(textoOtra, { line: linea, ch: 0 }, { line: linea, ch: textoActual.length });
      cm.replaceRange(textoActual, { line: otraLinea, ch: 0 }, { line: otraLinea, ch: textoOtra.length });

      cm.setCursor({ line: otraLinea, ch: sel.head.ch });
    });
  });
}

function duplicarLinea(cm, direccion) {
  const cursor = cm.getCursor();
  const line = cursor.line;
  const textActual = cm.getLine(line);
  const destino = direccion > 0 ? line + 1 : line;

  cm.operation(() => {
    cm.replaceRange(textActual + "\n", { line: destino, ch: 0 });
    cm.setCursor({ line: destino, ch: cursor.ch });
    cm.focus(); // <- mantener el foco
  });
}

function agregarPrototiposCpp(codigo) {
  const lineas = codigo.split('\n');
  const regexFuncion = /^(\w[\w\s:*&<>]*?)\s+(\w+)\s*\(([^)]*)\)\s*{/;
  const regexPrototipo = /^(\w[\w\s:*&<>]*?)\s+(\w+)\s*\(([^)]*)\)\s*;/;
  const funciones = [];
  let indiceMain = -1;

  for (let i = 0; i < lineas.length; i++) {
    if (/int\s+main\s*\(/.test(lineas[i])) {
      indiceMain = i;
      break;
    }
  }

  if (indiceMain === -1) {
    alert("No se encontró la función main().");
    return codigo;
  }

  for (let i = 0; i < lineas.length; i++) {
    const linea = lineas[i].trim();
    const match = linea.match(regexFuncion);
    if (match) {
      const tipoRetorno = match[1].trim();
      const nombreFuncion = match[2].trim();
      const parametros = match[3].trim();

      if (nombreFuncion === 'main') continue;

      const tiposParams = parametros
        ? parametros
            .split(',')
            .map(p => {
              let partes = p.trim().split(/\s+/);
              let tipo = partes.slice(0, -1).join(' ');
              let nombre = partes[partes.length - 1] || '';
              let arrayMatch = nombre.match(/(\[.*\])/);
              if (arrayMatch) {
                tipo += arrayMatch[1];
              }
              if (partes.length === 1) {
                tipo = partes[0];
              }
              return tipo;
            })
            .join(', ')
        : '';

      funciones.push(`${tipoRetorno} ${nombreFuncion}(${tiposParams});`);
    }
  }

  function normalizarPrototipo(str) {
    return str.replace(/\s+/g, '').replace(/;$/, '');
  }
  const prototiposExistentes = new Set();
  for (let i = 0; i < indiceMain; i++) {
    const linea = lineas[i].trim();
    if (regexPrototipo.test(linea)) {
      prototiposExistentes.add(normalizarPrototipo(linea));
    }
  }

  const prototiposFinales = funciones.filter(p => !prototiposExistentes.has(normalizarPrototipo(p)));

  if (prototiposFinales.length > 0) {
    let ultimaLineaPrototipo = -1;
    for (let i = 0; i < indiceMain; i++) {
      if (regexPrototipo.test(lineas[i].trim())) {
        ultimaLineaPrototipo = i;
      }
    }
    let nuevasLineas;
    if (ultimaLineaPrototipo >= 0) {
      nuevasLineas = [
        ...lineas.slice(0, ultimaLineaPrototipo + 1),
        ...prototiposFinales,
        ...lineas.slice(ultimaLineaPrototipo + 1)
      ];
    } else {
      nuevasLineas = [
        ...lineas.slice(0, indiceMain),
        ...prototiposFinales,
        '',
        ...lineas.slice(indiceMain)
      ];
    }
    return nuevasLineas.join('\n');
  }

  return codigo;
}

        urlsToTry.push(serverURLFromOrigin);
        console.log('🌐 Detectada URL de red desde window.location.origin:', serverURLFromOrigin);
      }
      
      if (currentHostname === 'localhost' || currentHostname === '127.0.0.1') {
        urlsToTry.push('http://localhost:3002');
        console.log('🏠 Usando localhost para conexión');
      }
      
      if (typeof getServerURLs === 'function') {
        const configUrls = getServerURLs();
        configUrls.forEach(url => {
          if (!urlsToTry.includes(url)) {
            urlsToTry.push(url);
          }
        });
      } else if (typeof getServerURL === 'function') {
        const configUrl = getServerURL();
        if (configUrl && !urlsToTry.includes(configUrl)) {
          urlsToTry.push(configUrl);
        }
      }
      
      if (urlsToTry.length === 0) {
        urlsToTry.push('http://localhost:3002');
      }
      
      console.log('🔗 URLs a intentar:', urlsToTry);
      
      intentarConectar(urlsToTry, 0, correoUsuario);
    }

    function intentarConectar(urls, index, correoUsuario) {
      if (index >= urls.length) {
        console.error('❌ No se pudo conectar a ningún servidor después de intentar todas las URLs');
        const urlsIntentadas = urls.join('\n- ');
        alert(`No se pudo conectar al servidor.\n\nURLs intentadas:\n- ${urlsIntentadas}\n\nAsegúrate de:\n1. Que el servidor esté corriendo\n2. Que el firewall permita conexiones en el puerto 3002\n3. Que estés usando la IP correcta del servidor`);
        socketListenersRegistrados = false;
        return;
      }

      if (socket) {
        socket.removeAllListeners();
        socket.disconnect();
        socket = null;
      }

      const serverURL = urls[index];
      console.log(`🔌 Intentando conectar a servidor ${index + 1}/${urls.length}: ${serverURL}`);
      
      socket = io(serverURL, {
        timeout: 10000, // Timeout de 10 segundos (aumentado)
        reconnection: false, // Desactivar reconexión automática para poder intentar otra URL
        transports: ['websocket', 'polling'], // Intentar websocket primero, luego polling como fallback
        upgrade: true, // Permitir upgrade de polling a websocket
        rememberUpgrade: false
      });

      let connectionEstablished = false;

      const connectionTimeout = setTimeout(() => {
        if (!connectionEstablished && socket && !socket.connected) {
          console.log(`⏱️ Timeout: Conexión a ${serverURL} no respondió en 10 segundos, intentando siguiente URL...`);
          socket.removeAllListeners();
          socket.disconnect();
          socket = null;
          intentarConectar(urls, index + 1, correoUsuario);
        }
      }, 10000);

      socket.on('connect', () => {
        connectionEstablished = true;
        clearTimeout(connectionTimeout);
        console.log('✅ ¡Conectado exitosamente al servidor:', serverURL, '!');
        
        if (!socketListenersRegistrados) {
          registrarSocketListeners(correoUsuario);
          socketListenersRegistrados = true;
        }
        
        iniciarVerificacionConexion();
        
        const archivoGuardado = obtenerArchivoSeleccionado();
        const nombreArchivoParaEnviar = nombreArchivoActual || (archivoGuardado ? archivoGuardado.nombreArchivo : null);
        
        socket.emit('usuario-join', { 
          username: correoUsuario,
          nombreArchivo: nombreArchivoParaEnviar
        });
      });

      socket.on('connect_error', (error) => {
        if (connectionEstablished) return; // Ignorar errores si ya se estableció la conexión
        
        clearTimeout(connectionTimeout);
        console.log(`❌ Error conectando a ${serverURL}:`, error.message);
        console.log('   Tipo de error:', error.type);
        
        socket.removeAllListeners();
        socket.disconnect();
        socket = null;
        
        if (index + 1 >= urls.length) {
          socketListenersRegistrados = false;
        } else {
          setTimeout(() => {
            intentarConectar(urls, index + 1, correoUsuario);
          }, 500);
        }
      });

      socket.on('disconnect', (reason) => {
        if (!connectionEstablished) return;
        console.log('🔌 Desconectado del servidor. Razón:', reason);
        if (vaciarEditoresPorDesconexion) {
          vaciarEditoresPorDesconexion();
        }
      });
    }

    }

                sessionStorage.setItem("indice", indice);
                console.log(`📚 Índice de ejercicio cargado desde servidor: ${indice} para archivo: ${nombreArchivoDelIndice}`);
                if (document.querySelector('button[onclick="anterior()"]')) {
                  mostrarEjercicio();
                }

            } else {
              sessionStorage.setItem("indiceServidor", indiceGuardado);
              sessionStorage.setItem("nombreArchivoIndiceServidor", nombreArchivoDelIndice);
              console.log(`📚 Índice de ejercicio guardado para cuando se carguen los ejercicios: ${indiceGuardado} para archivo: ${nombreArchivoDelIndice}`);
            }
          } else if (!nombreArchivoDelIndice || !archivoActual) {
            if (typeof ejercicios !== 'undefined' && ejercicios.length > 0) {
              if (indiceGuardado >= 0 && indiceGuardado < ejercicios.length) {
                indice = indiceGuardado;
                sessionStorage.setItem("indice", indice);
                console.log(`📚 Índice de ejercicio cargado desde servidor (sin verificación de archivo): ${indice}`);

                  mostrarEjercicio();

              console.log(`📚 Índice de ejercicio guardado (sin verificación de archivo): ${indiceGuardado}`);

          clearInterval(intervaloReconexion);
          intervaloReconexion = null;
        }
        
        if (estabaReconectando) {
          console.log('✅ Reconexión exitosa. Recargando página...');
          setTimeout(() => {
            window.location.reload();

        }

        document.getElementById('usuarios-conectados').style.display = 'block';

          minimizarBtn.addEventListener('click', toggleMenuUsuarios);
          minimizarBtn.setAttribute('data-listener-added', 'true');
        }

        if (toggleCursos && !toggleCursos.hasAttribute('data-listener-added')) {
          toggleCursos.addEventListener('click', toggleCursosRemotos);
          toggleCursos.setAttribute('data-listener-added', 'true');
        }

        if (toggleColaboracion && !toggleColaboracion.hasAttribute('data-listener-added')) {
          toggleColaboracion.addEventListener('click', toggleColaboracion);
          toggleColaboracion.setAttribute('data-listener-added', 'true');
          if (permitirColaboracion) {
            toggleColaboracion.classList.add('active');
          } else {
            toggleColaboracion.classList.remove('active');
          }
        }

          data.documentos.forEach(doc => {
            documentosDisponibles.set(doc.tabId, doc);
            crearPestañaUI(doc.tabId, doc.nombre, false);
          });
          actualizarEstadoVisualPestañas();
          if (data.documentos.length > 0) {
            cambiarPestaña(data.documentos[0].tabId);
          }
        } else {
          crearNuevaPestaña();
        }
        
        setTimeout(() => {
          const editorWasOpen = localStorage.getItem('editorAbierto') === 'true';
          if (editorWasOpen) {
            const overlay = document.getElementById("editor-overlay");
            const editorBtn = document.getElementById("abrir-editor");
            if (overlay && overlay.style.display !== "flex") {
              toggleEditor();
            }

      });

      socket.on('pestaña-creada', (data) => {
        documentosDisponibles.set(data.tabId, { nombre: data.nombre, codigo: data.codigo, creadorSocketId: data.creadorSocketId });
        crearPestañaUI(data.tabId, data.nombre, false);
        actualizarEstadoVisualPestañas();
        if (data.creadorSocketId === socket.id) {
          setTimeout(() => {
            cambiarPestaña(data.tabId);
          }, 100);
        }
      });

      socket.on('codigo-actual', (data) => {
        const { tabId, codigo } = data;
        const codigoServidor = codigo || '';
        const editorData = editoresPorPestaña.get(tabId);
        
        if (editorData && editorData.editorInstance) {
          const codigoActual = editorData.editorInstance.getValue();
          
          if (codigoServidor.trim() !== '' && codigoServidor !== codigoActual) {
            sincronizando = true;
            editorData.editorInstance.setValue(codigoServidor);
            editorData.codigo = codigoServidor;
          } else if (codigoServidor.trim() === '' && (codigoActual.trim() === '' || codigoActual === PLANTILLA_CPP || codigoActual === PLANTILLA_VECTOR || codigoActual === PLANTILLA_MATRIZ || codigoActual === PLANTILLA_LISTA_ENLAZADA)) {
            sincronizando = true;
            editorData.editorInstance.setValue(obtenerPlantillaActual());
            editorData.codigo = editorData.editorInstance.getValue();
          }
          
          if (codigoGuardadoAntesDesconexion.has(tabId)) {
            const codigoGuardado = codigoGuardadoAntesDesconexion.get(tabId);
            if (codigoGuardado && codigoGuardado.trim() !== '') {
              setTimeout(() => {
                if (editorData && editorData.editorInstance && socket && socket.connected) {
                  sincronizando = true;
                  editorData.editorInstance.setValue(codigoGuardado);
                  editorData.codigo = codigoGuardado;
                  console.log(`✅ Código restaurado para pestaña ${tabId} después de reconexión`);

                    const ultimaLinea = lineasCodigoGuardado.length - 1;

                      cambios: {
                        codigo: codigoGuardado,
                        from: { line: 0, ch: 0 },
                        to: { line: ultimaLinea, ch: ultimoChar },

                      usuario: miUsuario,
                      tabId: tabId
                    });
                  }
                  
                  codigoGuardadoAntesDesconexion.delete(tabId);
                }
              }, 500);
            }

        } else if (!editorData) {
          const docExistente = documentosDisponibles.get(tabId) || {};
          documentosDisponibles.set(tabId, { 
            ...docExistente, 
            codigo: codigoServidor,
            creadorSocketId: docExistente.creadorSocketId
          });

        const editorData = editoresPorPestaña.get(tabId);
        if (!editorData || !editorData.editorInstance) return;

        let nuevoCursor = cursor;
        if (lineaInsercion !== null && lineasInsertadas > 0 && cursor.line > lineaInsercion) {

        }

        editorData.editorInstance.setCursor(nuevoCursor);
        editorData.codigo = nuevoCodigo;

            editorData.cursosRemotos.forEach((curso, socketId) => {

                if (curso.line < editorData.editorInstance.lineCount()) {

                  const ch = Math.min(curso.ch, maxCh);

                  eliminarCursoRemotoEnPestaña(tabId, socketId);

              }
            });
          }
        }, 100);

      });

        setTimeout(() => {
        }, 100);
      });

        editoresPorPestaña.forEach((editorData, tabId) => {

        });

        const editorData = editoresPorPestaña.get(tabId);
        if (editorData && editorData.cursosRemotos) {

          if (!cursoExistente) {
            editorData.cursosRemotos.set(data.socketId, {
              marker: null,
              username: data.usuario,
              color: data.color,
              line: data.line,
              ch: data.ch

            cursoExistente.color = data.color;

        if (mostrarCursosRemotos) {

        if (data.valida) {

          if (toggle) {
            toggle.classList.add('active');
          }

        const editorCerrado = !overlay || overlay.style.display !== "flex";
        
        if (editorCerrado) {

            navegarAPosicionInvitada(tabId, line, ch, usuario, color);
          }, 200);

      socket.on('disconnect', () => {

        if (vaciarEditoresPorDesconexion) {
          vaciarEditoresPorDesconexion();
        }

      vaciarEditoresPorDesconexion = function() {
        limpiarEjercicios();
        
        mostrarIndicadorSincronizacion();
        
        const modalReconexion = document.getElementById('modal-reconexion');
        if (modalReconexion) {
          modalReconexion.style.display = 'flex';

          clearInterval(intervaloReconexion);
          intervaloReconexion = null;
        }
        
        if (miUsuario) {
          intervaloReconexion = setInterval(() => {
            const modal = document.getElementById('modal-reconexion');
            if (modal && modal.style.display === 'flex' && (!socket || !socket.connected)) {
              console.log('🔄 Intentando reconectar automáticamente...');
              const correoUsuario = obtenerCorreoUsuario() || miUsuario;
              if (correoUsuario) {
                const inputUsuario = document.getElementById('input-usuario');
                if (inputUsuario) {
                  inputUsuario.value = correoUsuario;
                }
                conectarUsuario();
              }
            } else if (socket && socket.connected) {
              clearInterval(intervaloReconexion);
              intervaloReconexion = null;
            }
          }, 3000); // Intentar cada 3 segundos
        }
        
        codigoGuardadoAntesDesconexion.clear();
        editoresPorPestaña.forEach((editorData, tabId) => {
          if (editorData && editorData.editorInstance) {
            const codigoActual = editorData.editorInstance.getValue();
            if (codigoActual && 
                codigoActual.trim() !== '' && 
                codigoActual !== PLANTILLA_CPP && 
                codigoActual !== PLANTILLA_VECTOR &&
                codigoActual !== PLANTILLA_MATRIZ &&
                codigoActual !== PLANTILLA_LISTA_ENLAZADA) {
              codigoGuardadoAntesDesconexion.set(tabId, codigoActual);
              console.log(`💾 Código guardado para pestaña ${tabId} antes de desconexión`);
            }
          }
        });
        
        editoresPorPestaña.forEach((editorData, tabId) => {
          if (editorData && editorData.editorInstance) {
            sincronizando = true; // Evitar que se sincronice con el servidor
            editorData.editorInstance.setValue(obtenerPlantillaActual());
            editorData.codigo = editorData.editorInstance.getValue();
            editorData.editorInstance.clearHistory();
            sincronizando = false;
            console.log(`🗑️ Editor de pestaña ${tabId} vaciado por desconexión (historial limpiado)`);
          }
        });
        
        socketListenersRegistrados = false;
        editoresPorPestaña.forEach((editorData, tabId) => {
          if (editorData.cursosRemotos) {
            editorData.cursosRemotos.forEach((curso, socketId) => {
              if (curso.marker) {
                curso.marker.clear();
              }
            });
            editorData.cursosRemotos.clear();
          }
        });
      }
    }

    
    function inicializarSistemaPestañas() {
      const nuevaPestañaBtn = document.getElementById('nueva-pestaña-btn');
      if (nuevaPestañaBtn) {
        nuevaPestañaBtn.addEventListener('click', crearNuevaPestaña);
      }

      document.addEventListener('keydown', function(e) {
        if (e.ctrlKey && e.key.toLowerCase() === 't') {
          e.preventDefault();
          const target = e.target;
          const isInputUsuario = target.id === 'input-usuario';
          
          const isInCodeMirror = target.closest && target.closest('.CodeMirror');
          
          if (!isInputUsuario || isInCodeMirror) {
            e.preventDefault();
            e.stopPropagation();
            crearNuevaPestaña();
          }
        }
      });

      const panelTemplate = document.querySelector('.editor-panel[data-tab-id=""]');
      if (panelTemplate) {
        panelTemplate.style.display = 'none';
        panelTemplate.classList.remove('active');
        panelTemplate.querySelectorAll('.input-area, .output, .editor-textarea').forEach(el => {
          el.style.display = 'none';
        });
      }
    }

    function crearNuevaPestaña() {
      if (!socket || !socket.connected) {
        alert('Debes estar conectado para crear una pestaña');
        return;
      }

      const nombre = prompt('Nombre de la nueva pestaña:') || `Pestaña ${editoresPorPestaña.size + 1}`;
      socket.emit('crear-pestaña', { nombre });
    }

    function crearPestañaUI(tabId, nombre, esActiva) {
      const tabsContainer = document.getElementById('editor-tabs');
      const nuevaPestañaBtn = document.getElementById('nueva-pestaña-btn');
      
      const tabBtn = document.createElement('button');
      tabBtn.className = 'editor-tab' + (esActiva ? ' active' : '');
      tabBtn.setAttribute('data-tab-id', tabId);
      tabBtn.innerHTML = `
        <span>${nombre}</span>
        <button class="editor-tab-close" data-tab-id="${tabId}">×</button>
      `;
      
      tabBtn.addEventListener('click', (e) => {
        if (e.target.classList.contains('editor-tab-close')) {
          e.stopPropagation();
          cerrarPestaña(tabId);

              e.stopPropagation();

            }
          }
          cambiarPestaña(tabId);
        }
      });

      tabsContainer.insertBefore(tabBtn, nuevaPestañaBtn);

      crearPanelEditor(tabId, nombre);

    function crearEditorCodeMirror(textarea, tabId) {
      if (textarea.cm) {
        return textarea.cm;
      }
      
      const parent = textarea.parentElement;
      if (parent) {
        const existingCM = parent.querySelector('.CodeMirror');
        if (existingCM && existingCM.CodeMirror) {
          textarea.cm = existingCM.CodeMirror;
          return existingCM.CodeMirror;
        }
      }
      
      const cppKeywords = [
        "if", "else", "while", "for", "do", "switch", "case", "break", "continue", "return",
        "struct", "class", "public", "private", "protected", "void", "int", "float", "double",
        "char", "bool", "true", "false", "namespace", "using", "include", "cout", "cin", "endl",
        "const", "static", "virtual", "override", "template", "typename", "try", "catch", "throw",
        "new", "delete", "this", "operator", "friend", "enum", "typedef", "sizeof", "main"
      ];

      function cppHint(cm) {
        const cursor = cm.getCursor();
        const token = cm.getTokenAt(cursor);
        const start = token.start;
        const end = cursor.ch;
        const currentWord = token.string.slice(0, end - start);
        let list = [];

        if (currentWord.length > 0) {
          list = cppKeywords.filter(k => k.startsWith(currentWord));
        } else {
          list = cppKeywords.slice();
        }

        const anyword = CodeMirror.hint.anyword(cm) || {list:[]};
        let all = [...new Set([...list, ...anyword.list])];

        return {
          list: all,
          from: CodeMirror.Pos(cursor.line, start),
          to: CodeMirror.Pos(cursor.line, end)
        };
      }

      const editor = CodeMirror.fromTextArea(textarea, {
        mode: "text/x-c++src",
        theme: "material",
        lineNumbers: true,
        indentUnit: 4,
        tabSize: 4,
        autoCloseBrackets: true,
        matchBrackets: true,
        styleActiveLine: true,
        foldGutter: true,
        gutters: ["CodeMirror-linenumbers", "CodeMirror-foldgutter"],
        highlightSelectionMatches: true,
        extraKeys: {
          "Shift-Space": "autocomplete",
          "Ctrl-D": function(cm) {
            let selections = cm.listSelections();
            let lastSel = selections[selections.length - 1];
            let selText = cm.getRange(lastSel.anchor, lastSel.head);
            let isEmpty = selText.length === 0;
            if (isEmpty) {
              let cursor = lastSel.head;
              let line = cm.getLine(cursor.line);
              let start = cursor.ch, end = cursor.ch;
              while (start > 0 && /\w/.test(line.charAt(start - 1))) start--;
              while (end < line.length && /\w/.test(line.charAt(end))) end++;
              if (start === end) return;
              cm.replaceSelection(line.slice(start, end), "around");
              selections = cm.listSelections();
              lastSel = selections[selections.length - 1];
            }
            let word = isEmpty ? selText : cm.getRange(lastSel.anchor, lastSel.head);
            let searchCursor = cm.getSearchCursor(word, lastSel.head);
            while (searchCursor.findNext()) {
              let alreadySelected = selections.some(sel =>
                CodeMirror.cmpPos(sel.anchor, searchCursor.from()) === 0 &&
                CodeMirror.cmpPos(sel.head, searchCursor.to()) === 0
              );
              if (!alreadySelected) {
                cm.addSelection(searchCursor.from(), searchCursor.to());
                break;
              }
            }
          }
        },
        hintOptions: { hint: cppHint }
      });

      const container = textarea.parentElement;
      if (container) {
        container.style.display = 'flex';
        container.style.flexDirection = 'column';
        container.style.height = '100%';
        container.style.minHeight = '500px';
      }

      const overlay = document.getElementById('editor-overlay');
      if (overlay && overlay.offsetHeight > 0) {
        const tabsHeight = document.querySelector('.editor-tabs')?.offsetHeight || 50;
        const headerHeight = textarea.closest('.editor-panel')?.querySelector('#editor-header')?.offsetHeight || 60;
        const availableHeight = overlay.offsetHeight - tabsHeight - headerHeight - 200; // espacio para input y output
        editor.setSize("100%", Math.max(400, availableHeight) + "px");
      } else {
        editor.setSize("100%", "500px");
      }
      
      setTimeout(() => {
        inicializarAutocompletadoPersonalizado(editor);
        inicializarSnippets(editor);
      }, 200);
      
      setTimeout(() => {
        editor.refresh();
        const cmDisplay = editor.getWrapperElement();
        if (cmDisplay) {
          cmDisplay.style.display = 'block';
          cmDisplay.style.height = '500px';
        }
      }, 100);

      const puedeEditar = puedeEditarPestaña(tabId);
      editor.setOption('readOnly', !puedeEditar);
      
      let cursorTimeout = null;
      
      editor.on('change', function(cm, change) {
        const editorData = editoresPorPestaña.get(tabId);
        if (!editorData) return;

        if (sincronizando) {
          sincronizando = false;
          return;
        }

        const nuevoCodigo = editor.getValue();
        editorData.codigo = nuevoCodigo;

        if (socket && socket.connected && miUsuario && tabId === pestañaActiva) {
          if (!puedeEditarPestaña(tabId)) {
            sincronizando = true;
            editor.setValue(editorData.codigo);
            sincronizando = false;
            return;
          }
          
          socket.emit('cambio-codigo', {
            cambios: { codigo: nuevoCodigo, from: change.from, to: change.to, text: change.text },
            usuario: miUsuario,
            tabId: tabId
          });
        }
      });

      editor.on('cursorActivity', function(cm) {
        if (socket && socket.connected && miUsuario && tabId === pestañaActiva) {
          if (cursorTimeout) clearTimeout(cursorTimeout);
          cursorTimeout = setTimeout(() => {
            const cursor = cm.getCursor();
            socket.emit('cursor-cambio', {
              usuario: miUsuario,
              color: miColor,
              line: cursor.line,
              ch: cursor.ch,
              tabId: tabId
            });
          }, 200);
        }
      });

      return editor;
    }

    function cambiarPestaña(tabId) {
      if (pestañaActiva === tabId) return;
      
      if (!permitirColaboracion && !navegandoPorInvitacion) {
        const docInfo = documentosDisponibles.get(tabId);
        if (docInfo && docInfo.creadorSocketId && docInfo.creadorSocketId !== socket.id) {
          alert('No puedes acceder a esta hoja. Solo puedes acceder a tus propias hojas cuando la colaboración está desactivada.');
          return;
        }
      }

      if (pestañaActiva) {
        const editorDataAnterior = editoresPorPestaña.get(pestañaActiva);
        if (editorDataAnterior) {
          const overlay = document.getElementById("editor-overlay");
          if (overlay) {
            editorDataAnterior.editorSoloMax = overlay.classList.contains("fullscreen-editor");
          }
        }
        
        const panelAnterior = document.querySelector(`.editor-panel[data-tab-id="${pestañaActiva}"]`);
        const tabAnterior = document.querySelector(`.editor-tab[data-tab-id="${pestañaActiva}"]`);
        if (panelAnterior) {
          panelAnterior.classList.remove('active');
          panelAnterior.style.display = 'none';
        }
        if (tabAnterior) tabAnterior.classList.remove('active');
      }

      const nuevoPanel = document.querySelector(`.editor-panel[data-tab-id="${tabId}"]`);
      const nuevoTab = document.querySelector(`.editor-tab[data-tab-id="${tabId}"]`);
      
      if (nuevoPanel) {
        nuevoPanel.classList.add('active');
        nuevoPanel.style.display = 'flex';
        
        const inputArea = nuevoPanel.querySelector('.input-area[data-tab-id="' + tabId + '"]');
        const output = nuevoPanel.querySelector('.output[data-tab-id="' + tabId + '"]');
        if (inputArea) {
          inputArea.style.display = 'block';
          inputArea.style.visibility = 'visible';
        }
        if (output) {
          output.style.display = 'block';
          output.style.visibility = 'visible';
          const outputContent = output.querySelector('.output-content');
          if (outputContent) {
            outputContent.style.display = 'block';
            outputContent.style.visibility = 'visible';
          }
        }
      }
      if (nuevoTab) nuevoTab.classList.add('active');

      const pestañaAnterior = pestañaActiva;
      
      pestañaActiva = tabId;
      
      const editorDataNueva = editoresPorPestaña.get(tabId);
      if (editorDataNueva) {
        const overlay = document.getElementById("editor-overlay");
        if (overlay && overlay.style.display === "flex") {
          if (editorDataNueva.editorSoloMax) {
            overlay.classList.add("fullscreen-editor");
            editorMaximizado = true;
          } else {
            overlay.classList.remove("fullscreen-editor");
            editorMaximizado = false;
          }
          
          const btnMaximizar = document.getElementById("maximizarSoloEditor");
          if (btnMaximizar) {
            if (editorDataNueva.editorSoloMax) {
              btnMaximizar.textContent = "🗗";
              btnMaximizar.title = "Restaurar vista completa";
            } else {
              btnMaximizar.textContent = "🗖";
              btnMaximizar.title = "Maximizar solo el editor";
            }
          }
        }
      }

      if (pestañaAnterior && pestañaAnterior !== tabId && socket && socket.connected) {
        const editorDataAnterior = editoresPorPestaña.get(pestañaAnterior);
        if (editorDataAnterior && editorDataAnterior.cursosRemotos) {
          editorDataAnterior.cursosRemotos.delete(socket.id);
          const cursoAnterior = editorDataAnterior.cursosRemotos.get(socket.id);
          if (cursoAnterior && cursoAnterior.marker) {
            cursoAnterior.marker.clear();
            cursoAnterior.marker = null;
          }
        }
      }

      editoresPorPestaña.forEach((data, otherTabId) => {
        if (otherTabId !== tabId && data.cursosRemotos) {
          data.cursosRemotos.forEach((curso, socketId) => {
            if (curso.marker) {
              curso.marker.clear();
              curso.marker = null;
            }
          });
          if (data.editorInstance) {
            data.editorInstance.refresh();
          }
        }
      });
      
      const editorDataActiva = editoresPorPestaña.get(tabId);
      if (editorDataActiva && editorDataActiva.cursosRemotos) {
        editorDataActiva.cursosRemotos.forEach((curso, socketId) => {
          if (curso.marker) {
            curso.marker.clear();
            curso.marker = null;
          }
        });
      }

      if (socket && socket.connected) {
        socket.emit('unirse-pestaña', { tabId });
        
        setTimeout(() => {
          const editorData = editoresPorPestaña.get(tabId);
          if (editorData && editorData.editorInstance && miUsuario) {
            const cursor = editorData.editorInstance.getCursor();
            socket.emit('cursor-cambio', {
              usuario: miUsuario,
              color: miColor,
              line: cursor.line,
              ch: cursor.ch,
              tabId: tabId
            });
          }
        }, 300);
      }

      let editorData = editoresPorPestaña.get(tabId);
      
      if (editorData && editorData.editorInstance) {
        const puedeEditar = puedeEditarPestaña(tabId);
        editorData.editorInstance.setOption('readOnly', !puedeEditar);
      }
      
      if (editorData && editorData.cursosRemotos && mostrarCursosRemotos && pestañaActiva === tabId) {
        setTimeout(() => {
          if (pestañaActiva === tabId) {
            editorData.cursosRemotos.forEach((curso, socketId) => {
              if (curso.line !== undefined && curso.ch !== undefined && curso.username && curso.color) {
                actualizarCursoRemotoEnPestaña(tabId, socketId, curso.username, curso.color, curso.line, curso.ch);
              }
            });
          }
        }, 150);
      }
      
      if (!editorData || !editorData.editorInstance) {
        const textarea = nuevoPanel?.querySelector('.editor-textarea[data-tab-id="' + tabId + '"]');
        if (textarea) {
          if (textarea.cm) {
            editorData = {
              editorInstance: textarea.cm,
              codigo: textarea.cm.getValue(),
              cursosRemotos: editoresPorPestaña.get(tabId)?.cursosRemotos || new Map(),
              nombre: documentosDisponibles.get(tabId)?.nombre || 'Pestaña',
              editorSoloMax: editoresPorPestaña.get(tabId)?.editorSoloMax || false
            };
            editoresPorPestaña.set(tabId, editorData);
          } else {
            textarea.id = `editor-${tabId}`;
            textarea.style.width = '100%';
            textarea.style.height = '500px';
            textarea.style.minHeight = '500px';
            
            const editorInstance = crearEditorCodeMirror(textarea, tabId);
            if (editorInstance) {
              const codigoDelDocumento = documentosDisponibles.get(tabId)?.codigo || '';
              const codigoInicial = editorInstance.getValue() || '';
              
              if (codigoDelDocumento.trim() !== '') {
                editorInstance.setValue(codigoDelDocumento);
              } else if (!codigoInicial || codigoInicial.trim() === '') {
                editorInstance.setValue(obtenerPlantillaActual());
              }
              
              if (!editorData) {
                editorData = {
                  editorInstance: editorInstance,
                  codigo: editorInstance.getValue(),
                  cursosRemotos: new Map(),
                  nombre: documentosDisponibles.get(tabId)?.nombre || 'Pestaña',
                  editorSoloMax: false
                };
              } else {
                editorData.editorInstance = editorInstance;
                editorData.codigo = editorInstance.getValue();
                if (editorData.editorSoloMax === undefined) {
                  editorData.editorSoloMax = false;
                }
              }
              editoresPorPestaña.set(tabId, editorData);
            }
          }
        }
      }
      
      if (editorData && editorData.editorInstance) {
        setTimeout(() => {
          const overlay = document.getElementById('editor-overlay');
          if (overlay && overlay.offsetHeight > 0) {
            const tabsHeight = document.querySelector('.editor-tabs')?.offsetHeight || 50;
            const headerHeight = nuevoPanel?.querySelector('[id^="editor-header"]')?.offsetHeight || 60;
            const availableHeight = overlay.offsetHeight - tabsHeight - headerHeight - 200;
            editorData.editorInstance.setSize("100%", Math.max(400, availableHeight) + "px");
          }
          
          editorData.editorInstance.refresh();
          editorData.editorInstance.focus();
          
          setTimeout(() => {
            editorData.editorInstance.refresh();
          }, 100);
        }, 200);
      }
    }

    function cerrarPestaña(tabId) {
      if (editoresPorPestaña.size <= 1) {
        alert('No puedes cerrar la última pestaña');
        return;
      }

      if (pestañaActiva === tabId) {
        const otrasPestañas = Array.from(editoresPorPestaña.keys()).filter(id => id !== tabId);
        if (otrasPestañas.length > 0) {
          cambiarPestaña(otrasPestañas[0]);
        }
      }

      const panel = document.querySelector(`.editor-panel[data-tab-id="${tabId}"]`);
      const tab = document.querySelector(`.editor-tab[data-tab-id="${tabId}"]`);
      
      if (panel) panel.remove();
      if (tab) tab.remove();

      const editorData = editoresPorPestaña.get(tabId);
      if (editorData && editorData.cursosRemotos) {
        editorData.cursosRemotos.forEach(curso => {
          if (curso.marker) curso.marker.clear();
        });
      }

      editoresPorPestaña.delete(tabId);
      documentosDisponibles.delete(tabId);
    }

    function actualizarCursoRemotoEnPestaña(tabId, socketId, username, color, line, ch) {
      const editorData = editoresPorPestaña.get(tabId);
      if (!editorData || !editorData.editorInstance) return;

      const editor = editorData.editorInstance;
      
      if (!editorData.cursosRemotos) {
        editorData.cursosRemotos = new Map();
      }

      if (editorData.cursosRemotos.has(socketId)) {
        const cursoExistente = editorData.cursosRemotos.get(socketId);
        if (cursoExistente.marker) {
          cursoExistente.marker.clear();
        }
      }

      const pos = CodeMirror.Pos(line, ch);
      const widget = document.createElement('span');
      widget.style.backgroundColor = color;
      widget.style.width = '2px';
      widget.style.height = '1.2em';
      widget.style.display = 'inline-block';
      widget.style.marginLeft = '-1px';
      widget.style.verticalAlign = 'text-bottom';

      const label = document.createElement('span');
      label.textContent = obtenerNombreUsuario(username);
      label.style.backgroundColor = color;
      label.style.color = 'white';
      label.style.position = 'absolute';
      label.style.top = '-1.5em';
      label.style.padding = '2px 6px';
      label.style.fontSize = '11px';
      label.style.fontWeight = 'bold';
      label.style.borderRadius = '3px';
      label.style.whiteSpace = 'nowrap';
      label.style.boxShadow = '0 1px 3px rgba(0,0,0,0.3)';
      label.style.zIndex = '1000';
      label.style.pointerEvents = 'none';

      const container = document.createElement('span');
      container.style.position = 'relative';
      container.style.display = 'inline-block';
      container.appendChild(widget);
      container.appendChild(label);

      const marker = editor.setBookmark(pos, {
        widget: container,
        insertLeft: false,
        handleMouseEvents: false
      });

      editorData.cursosRemotos.set(socketId, {
        marker: marker,
        username: username,
        color: color,
        line: line,
        ch: ch
      });

      setTimeout(() => editor.refresh(), 0);
    }

    function eliminarCursoRemotoEnPestaña(tabId, socketId) {
      const editorData = editoresPorPestaña.get(tabId);
      if (!editorData || !editorData.cursosRemotos) return;

      if (editorData.cursosRemotos.has(socketId)) {
        const curso = editorData.cursosRemotos.get(socketId);
        if (curso.marker) {
          curso.marker.clear();
        }
        editorData.cursosRemotos.delete(socketId);
        if (editorData.editorInstance) {
          setTimeout(() => editorData.editorInstance.refresh(), 0);
        }
      }
    }

    function obtenerNombreUsuario(correo) {
      if (!correo) return '';
      const partes = correo.split('@');
      return partes[0] || correo;
    }

      lista.innerHTML = '';
      
      usuariosPorSocketId.clear();

      usuarios.forEach(usuario => {

        });
        
        lista.appendChild(item);
      });
    }
    
    function invitarAMiPosicion(socketIdDestino) {
      if (!permitirColaboracion) {
        alert('La colaboración está desactivada. Activa "Permitir colaboración" para usar esta función.');
        return;
      }
      
      if (!socket || !socket.connected || !miUsuario) {
        alert('No estás conectado');
        return;
      }
      
      if (!pestañaActiva) {
        alert('No estás en ninguna pestaña activa');
        return;
      }
      
      const editorData = editoresPorPestaña.get(pestañaActiva);
      if (!editorData || !editorData.editorInstance) {
        alert('No hay editor activo');
        return;
      }
      
      const cursor = editorData.editorInstance.getCursor();
      
      socket.emit('invitar-a-mi-posicion', {

              curso.line !== null && curso.ch !== null) {

            if (editor.lineCount() === 0) {
              editor.focus();
              return;
            }
            
            const line = Math.max(0, Math.min(curso.line, editor.lineCount() - 1));
            const maxCh = Math.max(0, editor.getLine(line).length);
            const ch = Math.max(0, Math.min(curso.ch, maxCh));

            highlight.style.position = 'absolute';

            highlight.style.marginTop = '-1.2em';
            highlight.style.pointerEvents = 'none';
            highlight.style.transition = 'opacity 2s ease-out';
            
            const marker = editor.setBookmark(CodeMirror.Pos(line, 0), {
              widget: highlight,
              insertLeft: false
            });
            
            setTimeout(() => {
              if (highlight) {
                highlight.style.opacity = '0';
              }
            }, 100);
            
            setTimeout(() => {
              if (marker) marker.clear();
            }, 2000);
          } else {
            if (editorData.editorInstance) {
              editorData.editorInstance.focus();
            }
          }
        } else {
          if (editorData && editorData.editorInstance) {

    function inicializarArrastreMenu() {
      const menu = document.getElementById('usuarios-conectados');
      const header = document.getElementById('usuarios-header');
      const minimizarBtn = document.getElementById('minimizar-usuarios-btn');
      
      if (!menu || !header) return;

      menu.setAttribute('data-arrastre-inicializado', 'true');

      const posicionGuardada = localStorage.getItem('menuUsuariosPosicion');
      if (posicionGuardada) {
        try {
          const pos = JSON.parse(posicionGuardada);
          menu.style.top = pos.top + 'px';
          menu.style.left = pos.left + 'px';
          menu.style.transform = 'none'; // Quitar transform cuando hay posición guardada

          return;
        }

        menu.classList.add('arrastrando');
        
        const rect = menu.getBoundingClientRect();
        offsetMenu.x = e.clientX - rect.left;
        offsetMenu.y = e.clientY - rect.top;

        e.preventDefault();
      });

      document.addEventListener('mousemove', (e) => {
        if (!arrastrandoMenu) return;

        let nuevaX = e.clientX - offsetMenu.x;
        let nuevaY = e.clientY - offsetMenu.y;

        const rect = menu.getBoundingClientRect();
        const maxX = window.innerWidth - rect.width;
        const maxY = window.innerHeight - rect.height;

        nuevaX = Math.max(0, Math.min(nuevaX, maxX));
        nuevaY = Math.max(0, Math.min(nuevaY, maxY));

        menu.style.top = nuevaY + 'px';
        menu.style.left = nuevaX + 'px';
        menu.style.transition = 'none'; // Desactivar transición durante arrastre
      });

      document.addEventListener('mouseup', () => {
        if (arrastrandoMenu) {
          arrastrandoMenu = false;
          menu.classList.remove('arrastrando');
          menu.style.transition = 'box-shadow 0.3s ease'; // Reactivar transición

          menu.style.transform = 'none';
          const rect = menu.getBoundingClientRect();
          const posicion = {
            top: rect.top,
            left: rect.left
          };
          localStorage.setItem('menuUsuariosPosicion', JSON.stringify(posicion));
        }
      });
    }

    function selectAutocomplete(idx) {
      if (!autocompleteOptions.length) return;
      const word = autocompleteOptions[idx];
      const doc = editorInstance.getDoc();
      const cursor = doc.getCursor();
      const line = doc.getLine(cursor.line);
      const before = line.slice(0, cursor.ch);
      const match = before.match(/(\w+)$/);
      if (match) {
        const start = { line: cursor.line, ch: cursor.ch - match[1].length };
        doc.replaceRange(word, start, cursor);
      } else {
        doc.replaceRange(word, cursor);
      }
      hideAutocompleteMenu();
      editorInstance.focus();
    }

        const line = doc.getLine(cursor.line);
        const before = line.slice(0, cursor.ch);
        const match = before.match(/(\w+)$/);
        let prefix = match ? match[1] : '';
        let words = getWordsFromEditor().filter(w => w.startsWith(prefix) && w !== prefix);
        if (words.length === 0) {
          hideAutocompleteMenu();
          navegandoConShiftEspacio = false;
          return;
        }
        if (words.length === 1) {
          autocompleteOptions = words;
          autocompleteIndex = 0;
          autocompleteMenu.style.display = 'none'; // No mostrar menú
          selectAutocomplete(0); // Autocompletar inmediatamente
          navegandoConShiftEspacio = false;
          return;
        }
        if (autocompleteMenu.style.display === 'block' && navegandoConShiftEspacio) {
          autocompleteIndex = (autocompleteIndex + 1) % autocompleteOptions.length;
          showAutocompleteMenu(autocompleteOptions, autocompleteMenu.getBoundingClientRect());
        } else {
          autocompleteOptions = words;
          autocompleteIndex = 0;
          const coords = cm.cursorCoords(cursor, 'page');
          showAutocompleteMenu(words, coords);
          navegandoConShiftEspacio = true;
        }
      } else if (autocompleteMenu.style.display === 'block') {
        if (e.key === 'ArrowDown') {
          e.preventDefault();
          autocompleteIndex = (autocompleteIndex + 1) % autocompleteOptions.length;
          showAutocompleteMenu(autocompleteOptions, autocompleteMenu.getBoundingClientRect());
        } else if (e.key === 'ArrowUp') {
          e.preventDefault();
          autocompleteIndex = (autocompleteIndex - 1 + autocompleteOptions.length) % autocompleteOptions.length;
          showAutocompleteMenu(autocompleteOptions, autocompleteMenu.getBoundingClientRect());
        } else if (e.key === 'Enter') {
          e.preventDefault();
          selectAutocomplete(autocompleteIndex);
          navegandoConShiftEspacio = false;
        } else if (e.key === 'Escape') {
          hideAutocompleteMenu();
          navegandoConShiftEspacio = false;
        }
      }
    });

    snippetsInicializado.add(editorInstance);
    
    let snippetsMenu = document.getElementById('snippets-menu');
    if (!snippetsMenu) {
      snippetsMenu = document.createElement('div');
      snippetsMenu.id = 'snippets-menu';
      snippetsMenu.style.cssText = 'position: absolute; background: white; border: 1px solid #007bff; border-radius: 6px; box-shadow: 0 2px 8px rgba(0,0,0,0.2); z-index: 10001; display: none; max-height: 300px; overflow-y: auto; min-width: 250px;';
      document.body.appendChild(snippetsMenu);
    }
    
    let snippetsOptions = [];
    let snippetsIndex = 0;
    let searchQuery = '';
    let menuCoords = null; // Guardar las coordenadas originales del menú

    function filtrarSnippets(query) {
      if (!query || query.trim() === '') {
        return snippets;
      }
      const lowerQuery = query.toLowerCase();
      return snippets.filter(snippet => 
        snippet.nombre.toLowerCase().includes(lowerQuery)
      );
    }

    function showSnippetsMenu(filteredSnippets, coords, preservePosition = false) {
      snippetsMenu.innerHTML = '';
      
      if (!preservePosition && coords) {
        menuCoords = { left: coords.left, top: coords.top };
      }
      
      const finalCoords = preservePosition && menuCoords ? menuCoords : coords;
      
      if (filteredSnippets.length === 0) {
        const div = document.createElement('div');
        div.textContent = 'No se encontraron snippets';
        div.style.padding = '8px 12px';
        div.style.color = '#666';
        div.style.fontStyle = 'italic';
        snippetsMenu.appendChild(div);
      } else {
        filteredSnippets.forEach((snippet, i) => {
          const div = document.createElement('div');
          div.style.padding = '6px 12px';
          div.style.cursor = 'pointer';
          div.style.borderBottom = i < filteredSnippets.length - 1 ? '1px solid #eee' : 'none';
          div.style.background = (i === snippetsIndex) ? '#e3f2fd' : 'white';
          
          const nombreDiv = document.createElement('div');
          nombreDiv.textContent = snippet.nombre;
          nombreDiv.style.fontWeight = '500';
          nombreDiv.style.color = '#333';
          div.appendChild(nombreDiv);
          
          const previewDiv = document.createElement('div');
          const primeraLinea = snippet.codigo.split('\n')[0];
          previewDiv.textContent = primeraLinea.length > 40 ? primeraLinea.substring(0, 37) + '...' : primeraLinea;
          previewDiv.style.fontSize = '0.85em';
          previewDiv.style.color = '#666';
          previewDiv.style.marginTop = '2px';
          previewDiv.style.fontFamily = 'monospace';
          div.appendChild(previewDiv);
          
          div.onmouseenter = () => {
            snippetsIndex = i;
            if (menuCoords) {
              showSnippetsMenu(snippetsOptions, menuCoords, true);
            }
          };
          div.onclick = () => selectSnippet(i);
          snippetsMenu.appendChild(div);
        });
      }
      
      snippetsMenu.style.display = 'block';
      if (finalCoords) {
        snippetsMenu.style.left = finalCoords.left + 'px';
        snippetsMenu.style.top = (finalCoords.top + 20) + 'px';
      }
      
      if (filteredSnippets.length > 0 && snippetsIndex >= 0 && snippetsIndex < filteredSnippets.length) {
        const children = Array.from(snippetsMenu.children);
        const selectedElement = children[snippetsIndex];
        if (selectedElement) {
          selectedElement.scrollIntoView({
            behavior: 'smooth',
            block: 'nearest'
          });
        }
      }
    }

    function hideSnippetsMenu() {
      snippetsMenu.style.display = 'none';
      snippetsOptions = [];
      snippetsIndex = 0;
      searchQuery = '';
      menuCoords = null;
    }

    function selectSnippet(idx) {
      if (!snippetsOptions.length || idx < 0 || idx >= snippetsOptions.length) return;
      
      const snippet = snippetsOptions[idx];
      const doc = editorInstance.getDoc();
      const cursor = doc.getCursor();
      
      doc.replaceRange(snippet.codigo, cursor);
      
      const lines = snippet.codigo.split('\n');
      const newLine = cursor.line + lines.length - 1;
      const newCh = lines.length === 1 ? cursor.ch + snippet.codigo.length : lines[lines.length - 1].length;
      
      const tabMatch = snippet.codigo.match(/\n(\s*)/);
      if (tabMatch && tabMatch[1]) {
        const posTab = snippet.codigo.indexOf('\n' + tabMatch[1]);
        if (posTab !== -1) {
          const lineBeforeTab = snippet.codigo.substring(0, posTab).split('\n').length - 1;
          doc.setCursor({ line: cursor.line + lineBeforeTab, ch: tabMatch[1].length });
        } else {
          doc.setCursor({ line: newLine, ch: newCh });
        }
      } else {
        doc.setCursor({ line: newLine, ch: newCh });
      }
      
      hideSnippetsMenu();
      editorInstance.focus();
    }

    editorInstance.on('keydown', function(cm, e) {
      if (e.altKey && e.key.toLowerCase() === 'p') {
        e.preventDefault();
        const doc = cm.getDoc();
        const cursor = doc.getCursor();
        const coords = cm.cursorCoords(cursor, 'page');
        
        snippetsOptions = snippets;
        snippetsIndex = 0;
        searchQuery = '';
        showSnippetsMenu(snippetsOptions, coords);
        return;
      }
      
      if (snippetsMenu.style.display === 'block') {
        if (e.key === 'ArrowDown') {
          e.preventDefault();
          snippetsIndex = (snippetsIndex + 1) % snippetsOptions.length;
          if (menuCoords) {
            showSnippetsMenu(snippetsOptions, menuCoords, true);
          }
        } else if (e.key === 'ArrowUp') {
          e.preventDefault();
          snippetsIndex = (snippetsIndex - 1 + snippetsOptions.length) % snippetsOptions.length;
          if (menuCoords) {
            showSnippetsMenu(snippetsOptions, menuCoords, true);
          }
        } else if (e.key === 'Enter') {
          e.preventDefault();
          selectSnippet(snippetsIndex);
        } else if (e.key === 'Escape') {
          e.preventDefault();
          hideSnippetsMenu();
        } else if (e.key.length === 1 && !e.ctrlKey && !e.metaKey) {
          e.preventDefault();
          searchQuery += e.key;
          snippetsOptions = filtrarSnippets(searchQuery);
          snippetsIndex = 0;
          const doc = cm.getDoc();
          const cursor = doc.getCursor();
          const coords = cm.cursorCoords(cursor, 'page');
          showSnippetsMenu(snippetsOptions, coords);
        } else if (e.key === 'Backspace') {
          e.preventDefault();
          searchQuery = searchQuery.slice(0, -1);
          snippetsOptions = filtrarSnippets(searchQuery);
          snippetsIndex = 0;
          const doc = cm.getDoc();
          const cursor = doc.getCursor();
          const coords = cm.cursorCoords(cursor, 'page');
          showSnippetsMenu(snippetsOptions, coords);
        }
      }
    });

    document.addEventListener('mousedown', function(e) {
      if (snippetsMenu.style.display === 'block' && !snippetsMenu.contains(e.target)) {
        hideSnippetsMenu();
      }
    });
  }

  
  let portapapelesInterno = '';
  
  let usarPortapapelesSistema = false;
  let timestampPortapapelesSistema = 0;
  let accionPortapapelesSistema = ''; // 'copy', 'paste', 'cut', o ''

  function copiarConExecCommand(texto) {
    const textArea = document.createElement('textarea');
    textArea.value = texto;
    textArea.style.position = 'fixed';
    textArea.style.left = '-999999px';
    textArea.style.top = '-999999px';
    document.body.appendChild(textArea);
    textArea.select();
    try {
      document.execCommand('copy');
      console.log('Texto copiado al portapapeles del sistema usando execCommand');
    } catch (err) {
      console.error('Error al copiar con execCommand:', err);
    }
    document.body.removeChild(textArea);
  }

  function obtenerTextoSeleccionado(elemento) {
    if (elemento instanceof CodeMirror) {
      const seleccion = elemento.getSelection();
      return seleccion || '';
    } else if (elemento.tagName === 'TEXTAREA' || elemento.tagName === 'INPUT') {
      const inicio = elemento.selectionStart;
      const fin = elemento.selectionEnd;
      return elemento.value.substring(inicio, fin);
    } else {
      const seleccion = window.getSelection();
      return seleccion.toString();
    }
  }

  function insertarTexto(elemento, texto) {
    if (elemento instanceof CodeMirror) {
      const doc = elemento.getDoc();
      const seleccion = doc.getSelection();
      if (seleccion) {
        doc.replaceSelection(texto);
      } else {
        const cursor = doc.getCursor();
        doc.replaceRange(texto, cursor);
      }
    } else if (elemento.tagName === 'TEXTAREA' || elemento.tagName === 'INPUT') {
      const inicio = elemento.selectionStart;
      const fin = elemento.selectionEnd;
      const valorAnterior = elemento.value;
      const nuevoValor = valorAnterior.substring(0, inicio) + texto + valorAnterior.substring(fin);
      elemento.value = nuevoValor;
      const nuevaPosicion = inicio + texto.length;
      elemento.setSelectionRange(nuevaPosicion, nuevaPosicion);
      elemento.dispatchEvent(new Event('input', { bubbles: true }));
    }
  }

  function obtenerElementoActivo() {
    const elementoActivo = document.activeElement;
    if (elementoActivo && (elementoActivo.tagName === 'TEXTAREA' || elementoActivo.tagName === 'INPUT')) {
      if (elementoActivo.cm) {
        return elementoActivo.cm;
      }
      return elementoActivo;
    }

    if (typeof pestañaActiva !== 'undefined' && pestañaActiva) {
      const editorData = editoresPorPestaña.get(pestañaActiva);
      if (editorData && editorData.editorInstance) {
        const wrapper = editorData.editorInstance.getWrapperElement();
        if (wrapper && (wrapper.classList.contains('CodeMirror-focused') || wrapper.contains(document.activeElement))) {
          return editorData.editorInstance;
        }
      }
    }

    const editoresActivos = document.querySelectorAll('.CodeMirror-focused');
    if (editoresActivos.length > 0) {
      const cmElement = editoresActivos[0];
      if (cmElement && cmElement.CodeMirror) {
        return cmElement.CodeMirror;
      }
    }

    if (typeof editorInstance !== 'undefined' && editorInstance) {
      return editorInstance;
    }

    return null;
  }

  document.addEventListener('copy', function(e) {
    const tiempoTranscurrido = Date.now() - timestampPortapapelesSistema;
    console.log('Evento copy detectado. Bandera:', usarPortapapelesSistema, 'Acción:', accionPortapapelesSistema, 'Tiempo transcurrido:', tiempoTranscurrido);
    
    if (usarPortapapelesSistema && accionPortapapelesSistema === 'copy' && tiempoTranscurrido < 500) {
      console.log('Permitiendo copiado normal del sistema (Ctrl+Shift+C) - NO interceptando');
      usarPortapapelesSistema = false; // Resetear la bandera
      accionPortapapelesSistema = '';
      timestampPortapapelesSistema = 0;
      return;
    }
    
    const elemento = obtenerElementoActivo();
    if (!elemento) {
      console.log('No hay elemento activo en copy - permitiendo comportamiento normal');
      return;
    }

    const textoSeleccionado = obtenerTextoSeleccionado(elemento);
    
    if (textoSeleccionado) {
      portapapelesInterno = textoSeleccionado;
      
      e.preventDefault();
      e.stopPropagation();
      e.stopImmediatePropagation();
      
      if (e.clipboardData) {
        e.clipboardData.setData('text/plain', '');
      }
      
      console.log('Texto copiado al portapapeles interno (no se guardó en el sistema)');
    }
  }, true); // Usar capture phase para interceptar antes

  document.addEventListener('cut', function(e) {
    const tiempoTranscurrido = Date.now() - timestampPortapapelesSistema;
    console.log('Evento cut detectado. Bandera:', usarPortapapelesSistema, 'Acción:', accionPortapapelesSistema, 'Tiempo transcurrido:', tiempoTranscurrido);
    
    if (usarPortapapelesSistema && accionPortapapelesSistema === 'cut' && tiempoTranscurrido < 500) {
      console.log('Permitiendo cortado normal del sistema (Ctrl+Shift+X) - NO interceptando');
      usarPortapapelesSistema = false; // Resetear la bandera
      accionPortapapelesSistema = '';
      timestampPortapapelesSistema = 0;
      return;
    }
    
    const elemento = obtenerElementoActivo();
    if (!elemento) {
      console.log('No hay elemento activo en cut - permitiendo comportamiento normal');
      return;
    }

    const textoSeleccionado = obtenerTextoSeleccionado(elemento);
    
    if (textoSeleccionado) {
      portapapelesInterno = textoSeleccionado;
      
      if (elemento instanceof CodeMirror) {
        const doc = elemento.getDoc();
        doc.replaceSelection('');
      } else if (elemento.tagName === 'TEXTAREA' || elemento.tagName === 'INPUT') {
        const inicio = elemento.selectionStart;
        const fin = elemento.selectionEnd;
        elemento.value = elemento.value.substring(0, inicio) + elemento.value.substring(fin);
        elemento.setSelectionRange(inicio, inicio);
        elemento.dispatchEvent(new Event('input', { bubbles: true }));
      }
      
      e.preventDefault();
      e.stopPropagation();
      e.stopImmediatePropagation();
      
      if (e.clipboardData) {
        e.clipboardData.setData('text/plain', '');
      }
      
      console.log('Texto cortado y guardado en el portapapeles interno');
    }
  }, true);

  document.addEventListener('paste', function(e) {
    const tiempoTranscurrido = Date.now() - timestampPortapapelesSistema;
    console.log('Evento paste detectado. Bandera:', usarPortapapelesSistema, 'Acción:', accionPortapapelesSistema, 'Tiempo transcurrido:', tiempoTranscurrido);
    
    if (usarPortapapelesSistema && accionPortapapelesSistema === 'paste' && tiempoTranscurrido < 500) {
      console.log('Permitiendo pegado normal del sistema (Ctrl+Shift+V) - NO interceptando');
      usarPortapapelesSistema = false; // Resetear la bandera
      accionPortapapelesSistema = '';
      timestampPortapapelesSistema = 0;
      return;
    }
    
    const elemento = obtenerElementoActivo();
    if (!elemento) {
      console.log('No hay elemento activo - permitiendo comportamiento normal');
      return;
    }

    console.log('Interceptando pegado - usando portapapeles interno');
    e.preventDefault();
    e.stopPropagation();
    e.stopImmediatePropagation();
    
    if (portapapelesInterno) {
      insertarTexto(elemento, portapapelesInterno);
      console.log('Texto pegado desde el portapapeles interno');
    } else {
      console.log('El portapapeles interno está vacío - no se puede pegar');
    }
  }, true);

  document.addEventListener('keydown', function(e) {
    const tecla = e.key.toLowerCase();
    
    if ((e.ctrlKey || e.metaKey) && e.shiftKey && tecla === 'c' && !e.altKey) {
      e.preventDefault();
      e.stopPropagation();
      
      const elemento = obtenerElementoActivo();
      if (elemento) {
        const textoSeleccionado = obtenerTextoSeleccionado(elemento);
        if (textoSeleccionado) {
          console.log('Ctrl+Shift+C detectado - copiando al portapapeles del sistema');
          
          if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(textoSeleccionado).then(() => {
              console.log('Texto copiado al portapapeles del sistema exitosamente');
            }).catch((err) => {
              console.error('Error al copiar al portapapeles:', err);
              copiarConExecCommand(textoSeleccionado);
            });
          } else {
            copiarConExecCommand(textoSeleccionado);
          }
        } else {
          console.log('No hay texto seleccionado para copiar');
        }
      }
      return;
    }
    
    if ((e.ctrlKey || e.metaKey) && e.shiftKey && tecla === 'v' && !e.altKey) {
      usarPortapapelesSistema = true;
      accionPortapapelesSistema = 'paste';
      timestampPortapapelesSistema = Date.now();
      console.log('Ctrl+Shift+V detectado en keydown - usando portapapeles del sistema. Timestamp:', timestampPortapapelesSistema);
      return;
    }
    
    if ((e.ctrlKey || e.metaKey) && e.shiftKey && tecla === 'x' && !e.altKey) {
      usarPortapapelesSistema = true;
      accionPortapapelesSistema = 'cut';
      timestampPortapapelesSistema = Date.now();
      console.log('Ctrl+Shift+X detectado - usando portapapeles del sistema. Timestamp:', timestampPortapapelesSistema);
      return;
    }
    
    if ((e.ctrlKey || e.metaKey) && tecla === 'c' && !e.shiftKey && !e.altKey) {
      const elemento = obtenerElementoActivo();
      if (elemento) {
        const textoSeleccionado = obtenerTextoSeleccionado(elemento);
        if (textoSeleccionado) {
          portapapelesInterno = textoSeleccionado;
          e.preventDefault();
          e.stopPropagation();
          console.log('Ctrl+C interceptado - texto guardado en portapapeles interno');
        }
      }
    }
    
    if ((e.ctrlKey || e.metaKey) && tecla === 'x' && !e.shiftKey && !e.altKey) {
      const elemento = obtenerElementoActivo();
      if (elemento) {
        const textoSeleccionado = obtenerTextoSeleccionado(elemento);
        if (textoSeleccionado) {
          portapapelesInterno = textoSeleccionado;
          if (elemento instanceof CodeMirror) {
            const doc = elemento.getDoc();
            doc.replaceSelection('');
          } else if (elemento.tagName === 'TEXTAREA' || elemento.tagName === 'INPUT') {
            const inicio = elemento.selectionStart;
            const fin = elemento.selectionEnd;
            elemento.value = elemento.value.substring(0, inicio) + elemento.value.substring(fin);
            elemento.setSelectionRange(inicio, inicio);
            elemento.dispatchEvent(new Event('input', { bubbles: true }));
          }
          e.preventDefault();
          e.stopPropagation();
          console.log('Ctrl+X interceptado - texto cortado y guardado en portapapeles interno');
        }
      }
    }
    
    if ((e.ctrlKey || e.metaKey) && tecla === 'v' && !e.shiftKey && !e.altKey) {
      const elemento = obtenerElementoActivo();
      if (elemento) {
        if (portapapelesInterno) {
          e.preventDefault();
          e.stopPropagation();
          insertarTexto(elemento, portapapelesInterno);
          console.log('Ctrl+V interceptado - texto pegado desde portapapeles interno');
        } else {
        }
      }
    }
  }, true);
